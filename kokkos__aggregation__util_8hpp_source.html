71template <typename Agg_view_t>
72CPPUDDLE_HOST_DEVICE_METHOD typename Agg_view_t::view_type
73get_slice_subview(const size_t slice_id, const size_t max_slices,
74 const Agg_view_t &agg_view) {
75 const size_t slice_size = agg_view.size() / max_slices;
76 return Kokkos::subview(
77 agg_view, std::make_pair<size_t, size_t>(slice_id * slice_size,
78 (slice_id + 1) * slice_size));
79}

83template <typename Integer,
84 std::enable_if_t<std::is_integral<Integer>::value, bool> = true,
85 typename Agg_view_t, typename... Args>
86CPPUDDLE_HOST_DEVICE_METHOD auto
87map_views_to_slice(const Integer slice_id, const Integer max_slices,
88 const Agg_view_t &current_arg, const Args &...rest) {
89 static_assert(Kokkos::is_view<typename Agg_view_t::view_type>::value,
90 "Argument not an aggregated view");
91#if defined(HPX_COMPUTE_DEVICE_CODE) && defined(__NVCC__)
92 if constexpr (sizeof...(Args) > 0) {
93 return cuda::std::tuple_cat(
94 cuda::std::make_tuple(
95 get_slice_subview(slice_id, max_slices, current_arg)),
96 map_views_to_slice(slice_id, max_slices, rest...));
97 } else {
98 return cuda::std::make_tuple(
99 get_slice_subview(slice_id, max_slices, current_arg));
100 }
101#else
102 if constexpr (sizeof...(Args) > 0) {
103 return std::tuple_cat(
104 std::make_tuple(get_slice_subview(slice_id, max_slices, current_arg)),
105 map_views_to_slice(slice_id, max_slices, rest...));
106 } else {
107 return std::make_tuple(
108 get_slice_subview(slice_id, max_slices, current_arg));
109 }
110#endif
111}

115template <
116 typename Agg_executor_t, typename Agg_view_t,
117 std::enable_if_t<Kokkos::is_view<typename Agg_view_t::view_type>::value,
118 bool> = true,
119 typename... Args>
120CPPUDDLE_HOST_DEVICE_METHOD auto
121map_views_to_slice(const Agg_executor_t &agg_exec,
122 const Agg_view_t &current_arg, const Args &...rest) ->
123 std::tuple<typename Agg_view_t::view_type,
124 typename Args::view_type...> {
125 const size_t slice_id = agg_exec.id;
126 const size_t max_slices = agg_exec.max_slices;
127 static_assert(Kokkos::is_view<typename Agg_view_t::view_type>::value,
128 "Argument not an aggregated view");
129 if constexpr (sizeof...(Args) > 0) {
130 return std::tuple_cat(
131 std::make_tuple(get_slice_subview(slice_id, max_slices, current_arg)),
132 map_views_to_slice(agg_exec, rest...));
133 } else {
134 return std::make_tuple(
135 get_slice_subview(slice_id, max_slices, current_arg));
136 }
137}

140template <typename Agg_executor_t, typename TargetView_t, typename SourceView_t>
141void aggregated_deep_copy(Agg_executor_t &agg_exec, TargetView_t &target,
142 SourceView_t &source) {
143 if (agg_exec.sync_aggregation_slices()) {
144 Kokkos::deep_copy(agg_exec.get_underlying_executor().instance(), target,
145 source);

150template <typename Agg_executor_t, typename TargetView_t, typename SourceView_t>
151void aggregated_deep_copy(Agg_executor_t &agg_exec, TargetView_t &target,
152 SourceView_t &source, int elements_per_slice) {
153 if (agg_exec.sync_aggregation_slices()) {
154 const size_t number_slices = agg_exec.number_slices;
155 auto target_slices = Kokkos::subview(
156 target,
157 std::make_pair<size_t, size_t>(0, number_slices * elements_per_slice));
158 auto source_slices = Kokkos::subview(
159 source,
160 std::make_pair<size_t, size_t>(0, number_slices * elements_per_slice));
161 Kokkos::deep_copy(agg_exec.get_underlying_executor().instance(),
162 target_slices, source_slices);
163 }
164}

167template <typename executor_t, typename TargetView_t, typename SourceView_t>
168hpx::shared_future<void> aggregrated_deep_copy_async(
169 typename Aggregated_Executor<executor_t>::Executor_Slice &agg_exec,
170 TargetView_t &target, SourceView_t &source) {
171 const size_t gpu_id = agg_exec.parent.gpu_id;
172 auto launch_copy_lambda =
173 [gpu_id](TargetView_t &target, SourceView_t &source,
174 executor_t &exec) -> hpx::shared_future<void> {
175 cppuddle::executor_recycling::executor_pool::select_device<
176 executor_t, cppuddle::executor_recycling::round_robin_pool_impl<executor_t>>(gpu_id);
177 return hpx::kokkos::deep_copy_async(exec.instance(), target, source);
178 };
179 return agg_exec.wrap_async(launch_copy_lambda, target, source,
180 agg_exec.get_underlying_executor());
181}

184template <typename executor_t, typename TargetView_t, typename SourceView_t>
185hpx::shared_future<void> aggregrated_deep_copy_async(
186 typename Aggregated_Executor<executor_t>::Executor_Slice &agg_exec,
187 TargetView_t &target, SourceView_t &source, int elements_per_slice) {
188 const size_t number_slices = agg_exec.number_slices;
189 const size_t gpu_id = agg_exec.parent.gpu_id;
190 auto launch_copy_lambda = [gpu_id, elements_per_slice, number_slices](
191 TargetView_t &target, SourceView_t &source,
192 executor_t &exec) -> hpx::shared_future<void> {
193 cppuddle::executor_recycling::executor_pool::select_device<
194 executor_t,
195 cppuddle::executor_recycling::round_robin_pool_impl<executor_t>>(
196 gpu_id);
197 auto target_slices = Kokkos::subview(
198 target,
199 std::make_pair<size_t, size_t>(0, number_slices * elements_per_slice));
200 auto source_slices = Kokkos::subview(
201 source,
202 std::make_pair<size_t, size_t>(0, number_slices * elements_per_slice));
203 return hpx::kokkos::deep_copy_async(exec.instance(), target_slices,
204 source_slices);
205 };
206 return agg_exec.wrap_async(launch_copy_lambda, target, source,
207 agg_exec.get_underlying_executor());
208}
