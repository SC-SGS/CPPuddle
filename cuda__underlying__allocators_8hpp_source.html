32template <class T> struct cuda_pinned_allocator {
33 using value_type = T;
34 cuda_pinned_allocator() noexcept = default;
35 template <class U>
36 explicit cuda_pinned_allocator(cuda_pinned_allocator<U> const &) noexcept {}
37 T *allocate(std::size_t n) {
38 T *data;
39 cudaError_t error =
40 cudaMallocHost(reinterpret_cast<void **>(&data), n * sizeof(T));
41 if (error != cudaSuccess) {
42 std::string msg =
43 std::string(
44 "cuda_pinned_allocator failed due to cudaMallocHost failure : ") +
45 std::string(cudaGetErrorString(error));
46 throw std::runtime_error(msg);
47 }
48 return data;
49 }
50 void deallocate(T *p, std::size_t n) {
51 cudaError_t error = cudaFreeHost(p);
52 if (error != cudaSuccess) {
53 std::string msg =
54 std::string(
55 "cuda_pinned_allocator failed due to cudaFreeHost failure : ") +
56 std::string(cudaGetErrorString(error));

58 }
59 }
60};
